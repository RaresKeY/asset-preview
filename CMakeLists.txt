cmake_minimum_required(VERSION 3.22)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" ASSET_PREVIEW_VERSION LIMIT_COUNT 1)
project(asset_preview VERSION ${ASSET_PREVIEW_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network OpenGL)
find_package(f3d 4.0 QUIET COMPONENTS library)
if(NOT f3d_FOUND)
    find_package(f3d 3.5 REQUIRED COMPONENTS library)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
find_package(Threads REQUIRED)
find_package(KF6Parts QUIET)
# Lazy modules can be replaced while an older server is still running. Hash
# every shared C++ contract so it can reject an incompatible factory before any
# objects, callbacks or vtables cross the module boundary.
set(ASSET_PREVIEW_BACKEND_CONTRACTS
    src/viewer.h src/model_worker.h src/audio_waveform.h src/preview_cache.h CMakeLists.txt)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ASSET_PREVIEW_BACKEND_CONTRACTS})
string(TOUPPER "${CMAKE_BUILD_TYPE}" ASSET_PREVIEW_BUILD_TYPE)
set(ASSET_PREVIEW_BACKEND_CONTRACT "${CMAKE_CXX_COMPILER_ID};${CMAKE_CXX_COMPILER_VERSION};${CMAKE_CXX_STANDARD};${CMAKE_SIZEOF_VOID_P};${Qt6_VERSION};${CMAKE_CXX_FLAGS};${CMAKE_BUILD_TYPE};${CMAKE_CXX_FLAGS_${ASSET_PREVIEW_BUILD_TYPE}}")
foreach(contract IN LISTS ASSET_PREVIEW_BACKEND_CONTRACTS)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${contract}" contract_hash)
    string(APPEND ASSET_PREVIEW_BACKEND_CONTRACT ";${contract}:${contract_hash}")
endforeach()
string(SHA256 ASSET_PREVIEW_BACKEND_ABI "${ASSET_PREVIEW_BACKEND_CONTRACT}")
configure_file(src/backend_abi_config.h.in generated/backend_abi_config.h @ONLY)
add_executable(asset-preview-server src/main.cpp src/thumbnail_job.cpp src/crash_log.cpp src/service.cpp src/viewer.cpp src/model_surface.cpp src/preview_cache.cpp src/preview_preload.cpp src/cache_diagnostics.cpp src/overview_grid.cpp src/project_grid.cpp src/material_graph.cpp src/video_controls.cpp src/animation_controls.cpp src/fullscreen_notice.cpp
    src/inspection/document.cpp src/inspection/service.cpp src/inspection/panel.cpp
    src/codex_panel.cpp src/reference_editor.cpp)
target_link_libraries(asset-preview-server PRIVATE Qt6::Widgets Qt6::Network)
target_include_directories(asset-preview-server PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(asset-preview-server PRIVATE ASSET_PREVIEW_VERSION="${PROJECT_VERSION}")
if(KF6Parts_FOUND)
    target_link_libraries(asset-preview-server PRIVATE KF6::Parts)
    target_compile_definitions(asset-preview-server PRIVATE ASSET_PREVIEW_HAS_TERMINAL=1)
else()
    target_compile_definitions(asset-preview-server PRIVATE ASSET_PREVIEW_HAS_TERMINAL=0)
endif()
# The workstation's F3D dependency graph includes jemalloc with initial-exec TLS.
# It cannot be dlopened late. Map just that tiny dependency at startup, putting
# libstdc++ and libc first so C++ new/delete and C allocation share glibc's heap.
find_library(ASSET_PREVIEW_JEMALLOC jemalloc)
if(ASSET_PREVIEW_JEMALLOC AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_file(ASSET_PREVIEW_LIBC_DSO NAMES libc.so.6
        PATHS /usr/lib /lib /usr/lib/${CMAKE_LIBRARY_ARCHITECTURE} /lib/${CMAKE_LIBRARY_ARCHITECTURE} REQUIRED)
    find_file(ASSET_PREVIEW_STDCXX_DSO NAMES libstdc++.so.6
        PATHS /usr/lib /lib /usr/lib/${CMAKE_LIBRARY_ARCHITECTURE} /lib/${CMAKE_LIBRARY_ARCHITECTURE} REQUIRED)
    target_link_libraries(asset-preview-server PRIVATE "-Wl,--no-as-needed" ${ASSET_PREVIEW_STDCXX_DSO} ${ASSET_PREVIEW_LIBC_DSO} ${ASSET_PREVIEW_JEMALLOC} "-Wl,--as-needed")
endif()
target_compile_options(asset-preview-server PRIVATE -Wall -Wextra -Wpedantic -fno-omit-frame-pointer)
qt_add_resources(asset-preview-server preview-icon PREFIX "/" FILES icon.svg)
add_library(asset-preview-f3d MODULE src/model_view.cpp src/texture_preview.cpp src/gltf_import.cpp src/backend_abi.cpp)
target_include_directories(asset-preview-f3d PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_target_properties(asset-preview-f3d PROPERTIES PREFIX "" AUTOMOC OFF)
target_link_libraries(asset-preview-f3d PRIVATE Qt6::Widgets Qt6::OpenGL f3d::libf3d)
if(f3d_VERSION VERSION_GREATER_EQUAL 4.0)
    target_compile_definitions(asset-preview-f3d PRIVATE ASSET_PREVIEW_F3D4=1)
endif()
option(ASSET_PREVIEW_GLTF_BYTE_FILLS "Use native glTF strided-byte component fills when available" ON)
if(ASSET_PREVIEW_GLTF_BYTE_FILLS AND f3d_VERSION VERSION_GREATER_EQUAL 4.0)
    # The distro VTK CMake export has absent development targets. Resolve the
    # matching public SDK headers/libraries as the inspection adapter does.
    find_path(ASSET_PREVIEW_GLTF_INCLUDE vtkGLTFDocumentLoader.h PATH_SUFFIXES vtk vtk-9.7)
    get_target_property(ASSET_PREVIEW_F3D_INCLUDES f3d::libf3d INTERFACE_INCLUDE_DIRECTORIES)
    find_path(ASSET_PREVIEW_GLTF_SDK_INCLUDE f3d/vtk_ext/vtkF3DGLTFImporter.h
        HINTS ${ASSET_PREVIEW_F3D_INCLUDES})
    set(ASSET_PREVIEW_GLTF_READY TRUE)
    foreach(component IN ITEMS CommonCore CommonDataModel IOImport IOGeometry)
        find_library(ASSET_PREVIEW_GLTF_${component} NAMES vtk${component})
        if(NOT ASSET_PREVIEW_GLTF_${component})
            set(ASSET_PREVIEW_GLTF_READY FALSE)
        else()
            list(APPEND ASSET_PREVIEW_GLTF_LIBRARIES ${ASSET_PREVIEW_GLTF_${component}})
        endif()
    endforeach()
    if(ASSET_PREVIEW_GLTF_INCLUDE AND ASSET_PREVIEW_GLTF_READY AND ASSET_PREVIEW_GLTF_SDK_INCLUDE
        AND EXISTS "${ASSET_PREVIEW_GLTF_SDK_INCLUDE}/f3d/plugin.h"
        AND EXISTS "${ASSET_PREVIEW_GLTF_SDK_INCLUDE}/f3d/reader.h")
        target_compile_definitions(asset-preview-f3d PRIVATE ASSET_PREVIEW_GLTF_PIXELS=1)
        target_include_directories(asset-preview-f3d PRIVATE ${ASSET_PREVIEW_GLTF_INCLUDE} ${ASSET_PREVIEW_GLTF_SDK_INCLUDE})
        target_link_libraries(asset-preview-f3d PRIVATE ${ASSET_PREVIEW_GLTF_LIBRARIES})
    endif()
endif()
target_compile_options(asset-preview-f3d PRIVATE -Wall -Wextra -Wpedantic)

target_include_directories(asset-preview-mpv PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")

option(ASSET_PREVIEW_INSPECTION "Build the optional static glTF component inspector" ON)
set(ASSET_PREVIEW_INSPECTION_READY FALSE)
if(ASSET_PREVIEW_INSPECTION)
    # Some distro VTK exports reference absent header-only development targets.
    # This adapter uses public headers and existing shared libraries only. Keep
    # all VTK dependencies out of the server's startup link graph.
    find_path(ASSET_PREVIEW_VTK_INCLUDE vtkVersionMacros.h PATH_SUFFIXES vtk vtk-9.7)
    set(ASSET_PREVIEW_INSPECTION_READY TRUE)
    if(NOT ASSET_PREVIEW_VTK_INCLUDE)
        set(ASSET_PREVIEW_INSPECTION_READY FALSE)
    else()
        set(ASSET_PREVIEW_VTK_VERSION_HEADER "${ASSET_PREVIEW_VTK_INCLUDE}/vtkVersionMacros.h")
        if(EXISTS "${ASSET_PREVIEW_VTK_INCLUDE}/vtkVersionQuick.h")
            set(ASSET_PREVIEW_VTK_VERSION_HEADER "${ASSET_PREVIEW_VTK_INCLUDE}/vtkVersionQuick.h")
        endif()
        file(STRINGS "${ASSET_PREVIEW_VTK_VERSION_HEADER}" ASSET_PREVIEW_VTK_VERSION
            REGEX "^#define VTK_(MAJOR|MINOR)_VERSION[ \t]+")
        string(REGEX MATCH "VTK_MAJOR_VERSION[ \t]+([0-9]+)" _major "${ASSET_PREVIEW_VTK_VERSION}")
        set(ASSET_PREVIEW_VTK_MAJOR "${CMAKE_MATCH_1}")
        string(REGEX MATCH "VTK_MINOR_VERSION[ \t]+([0-9]+)" _minor "${ASSET_PREVIEW_VTK_VERSION}")
        set(ASSET_PREVIEW_VTK_MINOR "${CMAKE_MATCH_1}")
        if(NOT _major OR NOT _minor OR "${ASSET_PREVIEW_VTK_MAJOR}.${ASSET_PREVIEW_VTK_MINOR}" VERSION_LESS "9.7")
            set(ASSET_PREVIEW_INSPECTION_READY FALSE)
        endif()
    endif()
    foreach(component IN ITEMS IOImport RenderingExternal RenderingOpenGL2 RenderingCore FiltersSources CommonCore CommonDataModel CommonMath)
        find_library(ASSET_PREVIEW_VTK_${component} NAMES vtk${component})
        if(NOT ASSET_PREVIEW_VTK_${component})
            set(ASSET_PREVIEW_INSPECTION_READY FALSE)
        else()
            list(APPEND ASSET_PREVIEW_INSPECTION_LIBRARIES ${ASSET_PREVIEW_VTK_${component}})
        endif()
    endforeach()
    if(ASSET_PREVIEW_INSPECTION_READY)
        add_library(asset-preview-inspection MODULE src/inspection/vtk_view.cpp src/backend_abi.cpp)
        target_include_directories(asset-preview-inspection PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
        set_target_properties(asset-preview-inspection PROPERTIES PREFIX "" AUTOMOC OFF)
        target_include_directories(asset-preview-inspection PRIVATE ${ASSET_PREVIEW_VTK_INCLUDE})
        target_link_libraries(asset-preview-inspection PRIVATE Qt6::Widgets Qt6::OpenGL ${ASSET_PREVIEW_INSPECTION_LIBRARIES})
        target_compile_options(asset-preview-inspection PRIVATE -Wall -Wextra -Wpedantic)
    else()
        message(STATUS "Optional 3D inspection disabled: compatible VTK 9.7+ development files are unavailable")
    endif()
endif()
target_compile_definitions(asset-preview-server PRIVATE ASSET_PREVIEW_HAS_INSPECTION=$<BOOL:${ASSET_PREVIEW_INSPECTION_READY}>)
